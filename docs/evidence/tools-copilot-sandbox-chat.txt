2) Sandbox path validation logic

package edu.comp331.agent.tools;

import java.io.IOException;
import java.nio.file.*;
import java.util.Objects;

public final class Sandbox {
    private final Path root;

    public Sandbox(Path root) {
        this.root = Objects.requireNonNull(root, "root")
                .toAbsolutePath()
                .normalize();
    }

    public Path resolve(String userPath) {
        if (userPath == null || userPath.isBlank()) {
            throw new IllegalArgumentException("Path is required");
        }

        Path candidate = Path.of(userPath);
        Path normalized = candidate.isAbsolute()
                ? candidate
                : root.resolve(candidate).normalize();

        Path realRoot;
        try {
            realRoot = root.toRealPath(LinkOption.NOFOLLOW_LINKS);
        } catch (IOException e) {
            throw new IllegalArgumentException("Invalid sandbox root: " + root, e);
        }

        Path realCandidate;
        try {
            realCandidate = normalized.toRealPath(LinkOption.NOFOLLOW_LINKS);
        } catch (NoSuchFileException e) {
            // This allows a future file path as long as the normalized path itself stays inside root.
            realCandidate = normalized;
        } catch (IOException e) {
            throw new IllegalArgumentException("Invalid path: " + userPath, e);
        }

        if (!realCandidate.startsWith(realRoot)) {
            throw new IllegalArgumentException("Path escapes sandbox root: " + userPath);
        }

        return realCandidate;
    }
}

The key rules are:

null/blank input is rejected
relative paths are resolved against one root
absolute paths are only allowed if they still resolve under that root
.. traversal is rejected by startsWith(root)
symlinks are checked with toRealPath(...) so a link cannot escape the sandbox

void resolve_shouldAllow_readme_at_root() {
    Path root = Path.of("/tmp/sandbox");
    Sandbox sandbox = new Sandbox(root);

    assertEquals(root.resolve("README.md").normalize(), sandbox.resolve("README.md"));

@Test
void resolve_shouldBlock_parent_traversal() {
    Path root = Path.of("/tmp/sandbox");
    Sandbox sandbox = new Sandbox(root);

    assertThrows(IllegalArgumentException.class, () -> sandbox.resolve("../README.md"));
}
